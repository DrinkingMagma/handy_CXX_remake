213 void test_LineCodec_threadSafe() {
214  DEBUG("=== 开始LineCodec线程安全测试 ===");
215  LineCodec codec;
216  Buffer buf;
217  const int THREAD_NUM = 5;
218  const int LOOP_NUM = 100;
219 
220  // 线程函数：并发编码
221  auto encodeFunc = [&]() {
222  for (int i = 0; i < LOOP_NUM; ++i) {
223  std::string data = "thread_" + std::to_string(pthread_self()) + "_" + std::to_string(i);
224  codec.encode(data, buf);
225  g_codecTestCount.fetch_add(1, std::memory_order_relaxed);
226  }
227  };
228 
229  // 线程函数：并发解码
230  auto decodeFunc = [&]() {
231  for (int i = 0; i < LOOP_NUM; ++i) {
232  Slice msg;
233  size_t dataSize = buf.size();
234  if (dataSize > 0) {
235  codec.tryDecode(Slice(buf.data()), msg);
236  }
237  g_codecTestCount.fetch_add(1, std::memory_order_relaxed);
238  }
239  };
240 
241  // 启动线程（2个编码，3个解码）
242  std::vector<std::thread> threads;
243  for (int i = 0; i < 2; ++i) threads.emplace_back(encodeFunc);
244  for (int i = 0; i < 3; ++i) threads.emplace_back(decodeFunc);
245 
246  // 等待所有线程完成
247  for (auto& t : threads) t.join();
248 
249  // 验证执行次数（无崩溃即线程安全）
250  int total = g_codecTestCount.load();
251  int expected = THREAD_NUM * LOOP_NUM;
252  DEBUG("LineCodec线程安全测试: 总操作次数=%d（预期=%d），%s",
253  total, expected, total == expected ? "计数正确" : "计数异常");
