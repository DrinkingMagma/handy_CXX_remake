78  int LengthCodec::tryDecode(Slice data, Slice& msg)
79  {
80  std::lock_guard<std::mutex> lock(m_mutex);
81 
82  // 1. 检查头部是否为足够
83  if(data.size() < kHeaderLen)
84  return 0;
85 
86  // 2. 校验消息合法性
87  if(!checkMagic(data.data()))
88  return static_cast<int>(LengthCodec::DecodeErr::kInvalidMagic);
89 
90  // 3. 解析长度（网络字节序转换为主机字节序，避免类型转换溢出）
91  int32_t netLen = 0;
92  // 安全拷贝4字节长度（避免直接强转导致的内存对齐问题）
93  memcpy(&netLen, data.data() + 4, sizeof(netLen));
94  int32_t hostLen = Net::ntoh(netLen);
95 
96  // 4. 检查长度合法性（避免负数、0、超最大限制）
97  size_t maxLen = getMaxMsgLenUnSafe();
98  if(hostLen <= 0 || static_cast<size_t>(hostLen) > maxLen)
99  return static_cast<int>(LengthCodec::DecodeErr::kInvalidLength);
100 
101  // 5. 检查数据是否完整（总长度 = 头部长度 + 消息长度）
102  size_t totalNeeded = kHeaderLen + static_cast<size_t>(hostLen);
103  if(data.size() < totalNeeded)
104  return 0;
105 
106  // 6. 提取消息（确保Slice不越界）
107  msg = Slice(data.data() + kHeaderLen, static_cast<size_t>(hostLen));
108  return static_cast<int>(totalNeeded);
109  }
