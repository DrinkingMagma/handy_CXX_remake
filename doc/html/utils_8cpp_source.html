82  string Utils::format(const char* fmt, ...) noexcept
83  {
84  // 初始缓冲区大小（512字节，满足多数场景）
85  const size_t initialBufSize = 512;
86  // 最大缓冲区限制（1MB，防止恶意输入导致内存溢出）
87  const size_t maxBufSize = 1024 * 1024;
88 
89  // 初始化缓冲区（使用智能指针自动释放内存，防止内存泄漏）
90  auto buf = std::make_unique<char[]>(initialBufSize);
91  size_t currentBufSize = initialBufSize;
92 
93  while (true)
94  {
95  va_list ap;
96  // 初始化可变参数列表
97  va_start(ap, fmt);
98 
99  // vsnprintf：安全格式化，返回需要的缓冲区大小（不含终止符'\0'）
100  // 若返回值 < 0：格式化失败；若 >= currentBufSize：缓冲区不足
101  int neededSize = vsnprintf(buf.get(), currentBufSize, fmt, ap);
102  // 释放可变参数列表
103  va_end(ap);
104 
105  // 处理格式化失败（如非法占位符）
106  if(neededSize < 0)
107  return "";
108 
109  // 缓冲区足够：直接返回格式化结果
110  if(static_cast<size_t>(neededSize) < currentBufSize)
111  {
112  return string(buf.get(), static_cast<size_t>(neededSize));
113  }
114 
115  // 缓冲区不够：重新分配缓冲区，并重新格式化
116  // 计算新的缓冲区大小（+1用于存储终止符'\0'）
117  currentBufSize = static_cast<size_t>(neededSize) + 1;
118  // 若超出最大限制：拒绝扩容，避免内存耗尽
119  if (currentBufSize > maxBufSize)
120  return "";
121 
122  // 扩容缓冲区
123  buf.reset(new char[currentBufSize]);
124  }
125 
126  }

215  int Utils::addFdFlag(int fd, int flag) noexcept
216  {
217  // 校验文件描述符的合法性
218  if(fd < 0)
219  {
220  // 设置错误码：无效的文件描述符
221  errno = EBADF;
222  return -1;
223  }
224 
225  // 1. 获取当前文件描述符的标志
226  int currentFlags = fcntl(fd, F_GETFD);
227  if(currentFlags == -1)
228  {
229  // fcntl失败时，errno被系统自动设置
230  return -1;
231  }
232 
233  // 2. 检查标志是否已经存在（避免重复设置）
234  if((currentFlags & flag) == flag)
235  return 0;
236 
237  // 3. 添加标志并设置回文件描述符
238  if(fcntl(fd, F_SETFD, currentFlags | flag) == -1)
239  {
240  // fcntl失败时，errno被系统自动设置
241  return -1;
242  }
243 
244  return 0;
245  }

305  std::string Utils::getMimeType(const std::string& filename) noexcept
306  {
307  size_t dotPos = filename.find_first_of('.');
308  std::string ext;
309 
310  if(dotPos != std::string::npos && dotPos < filename.size() - 1)
311  {
312  // 提取扩展名并转换为小写
313  ext = filename.substr(dotPos);
314  std::transform(ext.begin(), ext.end(), ext.begin(), ::tolower);
315  }
316 
317  // 从映射表中查找，未找到则返回默认值
318  const auto& mimeMap = getMimeMap();
319  auto it = mimeMap.find(ext);
320  if(it != mimeMap.end())
321  return it->second;
322  // 未知类型默认二进制流
323  else
324  return "application/octet-stream";
