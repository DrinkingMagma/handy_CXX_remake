200  // 验证基本属性（避免优化掉代码）
201  if (s1.code() < 0 || s2.msg() == nullptr) {

206  }
207  };

215  for (auto& t : threads) {
216  t.join();
217  }
218 
219  // 验证总执行次数
220  int64_t total = g_statusThreadTestCount.load();
221  int64_t expected = THREAD_NUM * LOOP_NUM;
