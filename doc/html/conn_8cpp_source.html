255  void TcpConn::_handleRead(const TcpConnPtr& conn)
256  {
257  // 处理握手
258  if(getState() == State::HAND_SHAKING)
259  {
260  if(_handleHandshake(conn) != 0)
261  return;
262  }
263 
264  // 处理已连接状态的读事件
265  // 每次先扩展输入缓冲区，然后尝试读取数据至输入缓冲区末尾
266  while(getState() == State::CONNECTED)
267  {
268  m_inputBuffer.makeRoom();
269  int rd = 0;
270  int fd = -1;
271 
272  {
273  std::lock_guard<std::mutex> lock(m_ChannelMutex);
274  if(m_channel)
275  fd = m_channel->getFd();
276  }
277 
278  if(fd >= 0)
279  {
280  rd = _readImp(fd, m_inputBuffer.end(), m_inputBuffer.space());
281  TRACE("Channel %lld, fd %d, read %d bytes",
282  (long long)m_channel->getId(), fd, rd);
283  }
284  // 若被信号中断，继续读取
285  if(rd == -1 && errno == EINTR)
286  {
287  TRACE("Signal interruption, continue reading");
288  continue;
289  }
290  // 若没有数据可读，则结束循环
291  else if(rd == -1 && (errno == EAGAIN || errno == EWOULDBLOCK))
292  {
293  TRACE("No data to read, doing m_readCB(if exist) and break");
294  for(const auto& idleId : m_idleIds)
295  handleUpdateIdle(m_base, idleId);
296 
297  {
298  std::lock_guard<std::mutex> lock(m_callBacksMutex);
299  if(m_readCB && m_inputBuffer.size() > 0)
300  {
301  TRACE("Doing m_readCB");
302  m_readCB(conn);
303  }
304  }
305  break;
306  }
307  // 若连接关闭或出错
308  else if(fd == -1 || rd == 0 || rd == -1)
309  {
310  cleanup(conn);
311  break;
312  }
313  // rd大于0，即获取到了数据，需修改缓冲区的实际数据长度
314  else
315  {
316  m_inputBuffer.addSize(rd);
317  }
318  }
319  }

321  int TcpConn::_handleHandshake(const TcpConnPtr& conn)
322  {
323  FATAL_IF(getState() != State::HAND_SHAKING,
324  "handleHandShake called when state is not HandShaking, current state is %d",
325  static_cast<int>(getState()));
326 
327  int fd = -1;
328  {
329  std::lock_guard<std::mutex> lock(m_ChannelMutex);
330  if(m_channel)
331  fd = m_channel->getFd();
332  }
333 
334  if(fd < 0)
335  {
336  cleanup(conn);
337  return -1;
338  }
339 
340  // 以非阻塞方式监控fd的可写事件与错误事件
341  pollfd pFd;
342  pFd.fd = fd;
343  pFd.events = POLLOUT | POLLERR;
344  int r = poll(&pFd, 1, 0);
345 
346  // 若检测到可写事件
347  if(r == 1 && pFd.revents == POLLOUT)
348  {
349  {
350  std::lock_guard<std::mutex> lock(m_ChannelMutex);
351  if(m_channel)
352  m_channel->enableReadWrite(true, false);
353  }
354 
355  {
356  std::lock_guard<std::mutex> lock(m_stateMutex);
357  m_state = State::CONNECTED;
358  }
359  if(m_stateCB)
360  m_stateCB(conn);
361 
362  m_connectedTime_ms = Utils::timeMilli();
363  TRACE("TcpConn connected: %s -> %s, fd: %d",
364  m_local.toString().c_str(), m_peer.toString().c_str(), fd);
365  }
366  else
367  {
368  TRACE("Poll on fd %d returned %d, revents: %d", fd, r, pFd.revents);
369  cleanup(conn);
370  return -1;
371  }
372 
373  return 0;
374  }

376  void TcpConn::_handleWrite(const TcpConnPtr& conn)
377  {
378  State currentState = getState();
379 
380  if(currentState == State::HAND_SHAKING)
381  _handleHandshake(conn);
382  else if(currentState == State::CONNECTED)
383  {
384  ssize_t sended = _send(m_outputBuffer.begin(), m_outputBuffer.size());
385  m_outputBuffer.consume(sended);
386 
387  {
388  std::lock_guard<std::mutex> lock(m_callBacksMutex);
389  if(m_outputBuffer.empty() && m_writeCB)
390  m_writeCB(conn);
391  }
392 
393  bool isWritable = false;
394  {
395  std::lock_guard<std::mutex> lock(m_ChannelMutex);
396  if(m_channel)
397  isWritable = m_channel->isWritable();
398  }
399 
400  if(m_outputBuffer.empty() && isWritable)
401  {
402  // 写回调可能已经写入新的数据，因此需要检查是否仍然为空
403  if(m_outputBuffer.empty())
404  {
405  std::lock_guard<std::mutex> lock(m_ChannelMutex);
406  if(m_channel)
407  m_channel->enableWrite(false);
408  }
409  }
410  }
411  else
412  {
413  ERROR("Unexpected call to _handleWrite in state: %d", static_cast<int>(currentState));

417  ssize_t TcpConn::_send(const char* buf, size_t len)
418  {
419  if(len == 0)
420  return 0;
421 
422  size_t sended = 0;
423  int fd = -1;
424 
425  {
426  std::lock_guard<std::mutex> lock(m_ChannelMutex);
427  if(m_channel)
428  fd = m_channel->getFd();
429  }
430 
431  if(fd < 0)
432  return 0;
433 
434  while(len > sended)
435  {
436  ssize_t curWrited = _writeImp(fd, buf + sended, len - sended);
437  TRACE("Channel %lld, fd %d, wrote %ld bytes",
438  (long long)m_channel->getId(), fd, curWrited);
439 
440  if(curWrited > 0)
441  {
442  sended += curWrited;
443  continue;
444  }
445  // 被信号中断，继续写
446  else if(curWrited == -1 && errno == EINTR)
447  {
448  continue;
449  }
450  // 暂时无法写入，启用写事件监听
451  else if(curWrited == -1 && (errno == EAGAIN || errno == EWOULDBLOCK))
452  {
453  {
454  std::lock_guard<std::mutex> lock(m_ChannelMutex);
455  if(m_channel && !m_channel->isWritable())
456  m_channel->enableWrite(true);
457  }
458  break;
459  }
460  // 写入错误
461  else
462  {
463  ERROR("Write error on channel %lld, fd %d, curWrited %d, errno=%d, msg=%s",
464  (long long)m_channel->getId(), fd, (int)curWrited, errno, strerror(errno));
465  break;
466  }
467  }
468 
469  return sended;
470  }

472  void TcpConn::send(Buffer& buf)
473  {
474  if(buf.empty())
475  return;
476 
477  bool isChannelValid = false;
478  {
479  std::lock_guard<std::mutex> lock(m_ChannelMutex);
480  isChannelValid = (m_channel != nullptr);
481  }
482 
483  if(!isChannelValid)
484  {
485  WARN("Sending data to a closed connection: %s -> %s, %lu bytes lost",
486  m_local.toString().c_str(), m_peer.toString().c_str(), buf.size());
487  return;
488  }
489 
490  bool isWritable = false;
491  {
492  std::lock_guard<std::mutex> lock(m_ChannelMutex);
493  if(m_channel)
494  isWritable = m_channel->isWritable();
495  }
496 
497  if(isWritable)
498  {
499  // 若通道启用写事件，则将数据追加到输出缓冲区中
500  m_outputBuffer.absorb(buf);
501  }
502  // 尝试直接发送数据
503  else
504  {
505  ssize_t sended = _send(buf.begin(), buf.size());
506  buf.consume(sended);
507 
508  // 若还有未发送的数据，则追加到输出缓冲区中，并启用写事件监听
509  if(buf.size() > 0)
510  {
511  m_outputBuffer.absorb(buf);
512 
513  {
514  std::lock_guard<std::mutex> lock(m_ChannelMutex);
515  if(m_channel && !m_channel->isWritable())
516  m_channel->enableWrite(true);
517  }
518  }
519  }

556  {
557  std::lock_guard<std::mutex> lock(m_callBacksMutex);
558  FATAL_IF(m_readCB, "onMsg and onReadable are mutually exclusive");
559 
560  m_codec = std::move(codec);

562  {

564  while(r > 0)
565  {
566  Slice msg;

568  // 若解码错误，关闭连接

571  conn->m_channel->close();

573  }
574  else if(r > 0)
575  {
576  TRACE("Decoded a message. Original length: %d, message length: %ld",
577  r, msg.size());
578  cb(conn, msg);
579  conn->getInputBuffer().consume(r);
580  }

691  void TcpServer::_handleAccept()
692  {
693  int listenFd = -1;
694  {
695  std::lock_guard<std::mutex> lock(m_ChannelMutex);
696  if(m_listenChannel)
697  {
698  listenFd = m_listenChannel->getFd();
699  }
700  }
701 
702  if(listenFd < 0)
703  return;
704 
705  struct sockaddr_in remoteAddr;
706  socklen_t remoteSize = sizeof(remoteAddr);
707  int curFd;
708 
709  // 处理所有待接受的连接
710  while((curFd = accept(listenFd, (struct sockaddr*)&remoteAddr, &remoteSize)) >= 0)
711  {
712  // 获取对端地址
713  sockaddr_in local, peer;
714  socklen_t addrSize = sizeof(peer);
715  int r = getpeername(curFd, (struct sockaddr*)&peer, &addrSize);
716  if(r < 0)
717  {
718  close(curFd);
719  ERROR("getpeername failed: errno=%d, msg=%s", errno, strerror(errno));
720  continue;
