462  Buffer& Buffer::absorb(Buffer& other)
463  {
464  if(this == &other)
465  return *this;
466 
467  // 按锁地址排序，确保加锁顺序固定
468  std::mutex* lock1 = m_mutex.get();
469  std::mutex* lock2 = other.m_mutex.get();
470  if (lock1 > lock2) std::swap(lock1, lock2);
471 
472  std::lock_guard<std::mutex> lockA(*lock1);
473  std::lock_guard<std::mutex> lockB(*lock2);
474 
475  if(other.m_b == other.m_e)
476  return *this;
477 
478  if(m_b == m_e)
479  {
480  std::swap(m_buf, other.m_buf);
481  std::swap(m_cap, other.m_cap);
482  std::swap(m_b, other.m_b);
