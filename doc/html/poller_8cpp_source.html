217  int PollerEpoll::loopOnce(int waitTime_ms)
218  {
219  // 记录轮询开始时间
220  const int64_t startTime_ms = Utils::timeMilli();
221 
222  // 等待事件
223  m_lastActive = epoll_wait(m_epollFd, m_activeEvs, kMaxEvents, waitTime_ms);
224  const int64_t usedTime_ms = Utils::timeMilli() - startTime_ms;
225 
226  TRACE("poller.cpp::PollerEpoll::loopOnce(): PollerEpoll[%lld] epoll_wait, waitTime_ms=%d, m_lastActive=%d, usedTime_ms=%lld",
227  static_cast<long long>(getId()), waitTime_ms, m_lastActive, usedTime_ms);
228 
229  // 错误处理：排除信号中断（EINTR是正常情况）
230  if(m_lastActive < 0 && errno != EINTR)
231  {
232  throw std::runtime_error(
233  Utils::format("poller.cpp::PollerEpoll::loopOnce(): PollerEpoll[%lld] epoll_wait failed, waitTime_ms=%d, usedTime_ms=%d, errno=%d, msg=%s",
234  static_cast<long long>(getId()), waitTime_ms, usedTime_ms, errno, strerror(errno))
235  );
236  }
237 
238  // 处理活跃事件
239  for(int i = m_lastActive - 1; i >= 0; --i)
240  {
241  struct epoll_event& ev = m_activeEvs[i];
242  Channel* ch = static_cast<Channel*>(ev.data.ptr);
243 
244  if(ch == nullptr)
245  continue; // 已被移除的Channel，跳过
246 
247  // 处理读事件（包含错误事件POLLERR
248  if(ev.events & (kReadEvent | POLLERR))
249  {
250  TRACE("PollerEpoll::loopOnce(): PollerEpoll[%lld] handle read: Channel[%lld], fd=%d",
251  static_cast<long long>(getId()),
252  static_cast<long long>(ch->getId()),
253  ch->getFd());
254  ch->handleRead();
255  }
256  else if (ev.events & kWriteEvent)
257  {
258  TRACE("PollerEpoll::loopOnce(): PollerEpoll[%lld] handle write: Channel[%lld], fd=%d",
259  static_cast<long long>(getId()),
260  static_cast<long long>(ch->getId()),
261  ch->getFd());
262  ch->handleWrite();
263  }
264  else
265  {
266  FATAL("PollerEpoll::loopOnce(): PollerEpoll[%lld] unexpected event: Channel[%lld], fd=%d, events=0x%x",
267  static_cast<long long>(getId()),
268  static_cast<long long>(ch->getId()),
269  ch->getFd(),
270  ev.events);
271  throw std::runtime_error("poller.cpp::PollerEpoll::loopOnce(): unexpected event type");
272  }
273  }
274  return m_lastActive;
275  }

485  // 实现：执行一次 kqueue 轮询
486  int PollerKqueue::loopOnce(int wait_ms) {
487  // 初始化超时时间（kqueue 需 timespec 类型）
488  struct timespec timeout;
489  if (wait_ms < 0) {
490  // 无限等待：timespec 设为 NULL（通过特殊值标识）
491  timeout.tv_sec = 0;
492  timeout.tv_nsec = -1;
493  } else {
494  timeout.tv_sec = wait_ms / 1000;
495  timeout.tv_nsec = (wait_ms % 1000) * 1000 * 1000; // 转换为纳秒
496  }
497 
498  // 记录轮询开始时间（用于统计耗时）
499  const int64_t start_ms = Utils::timeMilli();
500 
501  // 调用 kevent 等待事件（根据 timeout 决定是否阻塞）
502  last_active_ = kevent(
503  m_kqueueFd,
504  nullptr, 0, // 无事件修改，仅等待
505  active_evs_, kMaxEvents, // 活跃事件存储
506  (wait_ms < 0) ? nullptr : &timeout // 超时参数
507  );
508  const int64_t used_ms = Utils::timeMilli() - start_ms;
509 
510  // 日志：输出轮询结果
511  trace("PollerKqueue[%lld] kevent: wait_ms=%d, return=%d, used_ms=%lld, errno=%d",
512  static_cast<long long>(getId()),
513  wait_ms, last_active_, used_ms, errno);
514 
515  // 错误处理：排除信号中断（EINTR 是正常情况，无需抛出异常）
516  if (last_active_ < 0 && errno != EINTR) {
517  throw std::runtime_error(
518  Utils::format("poller.cpp::PollerKqueue::loopOnce(): PollerKqueue[%lld] kevent failed: errno=%d, msg=%s",
519  static_cast<long long>(getId()), errno, strerror(errno)));
520  }
521 
522  // 处理活跃事件（从后往前遍历，避免删除元素影响索引）
523  for (int i = last_active_ - 1; i >= 0; --i) {
524  struct kevent& ev = active_evs_[i];
525  Channel* ch = static_cast<Channel*>(ev.udata);
526 
527  // 若 Channel 已被移除（udata 为空），跳过处理
528  if (ch == nullptr) {
529  continue;
530  }
531 
532  // 处理逻辑：优先处理写事件，其次处理读事件（含 EOF）
533  if (!(ev.flags & EV_EOF) && ch->writeEnabled()) {
534  trace("PollerKqueue[%lld] handle write: Channel[%lld], fd=%d",
535  static_cast<long long>(getId()),
536  static_cast<long long>(ch->id()),
537  ch->fd());
538  ch->handleWrite();
539  } else if ((ev.flags & EV_EOF) || ch->readEnabled()) {
540  trace("PollerKqueue[%lld] handle read: Channel[%lld], fd=%d",
541  static_cast<long long>(getId()),
542  static_cast<long long>(ch->id()),
543  ch->fd());
544  ch->handleRead();
545  }
546  // 异常事件：无预期的事件类型
547  else {
548  fatal("PollerKqueue[%lld] unexpected event: Channel[%lld], fd=%d, flags=0x%x",
549  static_cast<long long>(getId()),
550  static_cast<long long>(ch->id()),
551  ch->fd(), ev.flags);
552  throw std::runtime_error("poller.cpp::PollerKqueue::loopOnce(): unexpected event type");
553  }
554  }
555 
556  return last_active_; // 返回本次处理的活跃事件数量
557  }
