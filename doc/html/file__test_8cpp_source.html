237 void test_thread_safe() {
238  DEBUG("=== 开始测试文件操作线程安全 ===");
239 
240  const int THREAD_NUM = 4; // 4个测试线程
241  const int LOOP_NUM = 50; // 每个线程循环50次
242  const std::string baseFile = "thread_test_file_";
243  std::vector<std::thread> threads;
244 
245  // 线程函数：并发创建、写入、读取文件
246  auto thread_func = [&](int threadId) {
247  for (int i = 0; i < LOOP_NUM; ++i) {
248  // 每个线程操作不同的文件避免竞争
249  std::string filename = baseFile + std::to_string(threadId) + "_" + std::to_string(i);
250  std::string content = "线程" + std::to_string(threadId) + "循环" + std::to_string(i) + "的内容";
251 
252  // 写入文件
253  Status st = File::writeContent(filename, content);
254  if (st.ok()) {
255  // 读取文件验证
256  std::string readContent;
257  st = File::getContent(filename, readContent);
258  if (st.ok() && readContent == content) {
259  g_fileThreadTestCount.fetch_add(1, std::memory_order_relaxed);
260  }
261  }
262  // 清理临时文件
263  File::deleteFile(filename);
264  }
265  };
266 
267  // 创建线程
268  for (int i = 0; i < THREAD_NUM; ++i) {
269  threads.emplace_back(thread_func, i);
270  }
271 
272  // 等待所有线程结束
273  for (auto& t : threads) {
274  t.join();
275  }
276 
277  // 验证总成功次数
278  int64_t total = g_fileThreadTestCount.load();
279  int64_t expected = THREAD_NUM * LOOP_NUM;
280  bool threadOk = (total == expected);
281  DEBUG("线程安全测试：总成功次数=%lld（预期：%lld，%s）",
282  total, expected, threadOk ? "通过" : "失败");
283 
284  DEBUG("=== 文件操作线程安全测试结束 ===\n");
285 }
