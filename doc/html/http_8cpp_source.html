57  HttpMsg::Result HttpMsg::_tryDecode(Slice buf, bool isCopyBody, Slice* firstLine)
58  {
59  if(m_completed)
60  return Result::Complete;
61 
62  // 解析头部
63  if(m_contentLen == 0)
64  {
65  const char* data = buf.data();
66  size_t bufSize = buf.size();
67 
68  // 查找头部结束标记"\r\n\r\n"
69  while(m_scannedLen + 4 <= bufSize)
70  {
71  if(memcmp(data + m_scannedLen, "\r\n\r\n", 4) == 0)
72  {
73  // 提取头部内容（不含结束标记）
74  Slice header(data, m_scannedLen);
75  // 提取第一行（请求行/状态行）
76  *firstLine = header.eatLine();
77 
78  // 解析头部字段
79  while (header.size() > 0)
80  {
81  Slice line = header.eatLine();
82  // 若为空，说明头部解析结束
83  if(line.empty())
84  break;
85 
86  // 提取key（到":"为止)
87  Slice key = line.eatUntil(':');
88  if(key.empty() || line.empty())
89  {
90  ERROR("Invalid header line: %.*s", (int)line.size(), line.data());
91  return Result::Error;
92  }
93  // 移除':'
94  key = key.sub(0, key.size() - 1);
95  // 去除value的前后空格
96  line.trimSpace();
97 
98  // key转换为小写后存入头部映射
99  std::string lowerKey(key.data(), key.size());
100  for(char& c : lowerKey)
101  c = tolower(c);
102  m_headers[lowerKey] = std::string(line.data(), line.size());
103  }
104 
105  // 跳过头部结束标记
106  m_scannedLen += 4;
107  // 解析内容长度
108  std::string len = getHeaderValue("content-length");
109  if(len != "")
110  m_contentLen = stoull(len, nullptr, 10);
111 
112  // 检查是否需要发送100 Continue
113  if(bufSize < m_scannedLen + m_contentLen && !getHeaderValue("expect").empty())
114  return Result::Continue100;
115 
116  // 如果没有消息体（GET请求），则解析完成
117  if(m_contentLen == 0)
118  {
119  m_completed = true;
120  return Result::Complete;
121  }
122  break;
123  }
124  m_scannedLen++;
125  }
126  }
127 
128  // 解析消息体
129  if(buf.size() >= m_scannedLen + m_contentLen)
130  {
131  if(isCopyBody)
132  m_body.assign(buf.data() + m_scannedLen, m_contentLen);
133  else
134  m_body2 = Slice(buf.data() + m_scannedLen, m_contentLen);
135 
136  m_scannedLen += m_contentLen;
137  m_completed =true;
138  return Result::Complete;
139  }
140 
141  return Result::NotComplete;
142  }

170  HttpMsg::Result HttpRequest::tryDecode(Slice buf, bool isCopyBody)
171  {
172  Slice firstLine;
173  Result result = _tryDecode(buf, isCopyBody, &firstLine);
174 
175  if(!firstLine.empty())
176  {
177  // 解析请求行：方法URI版本
178  m_method = firstLine.eatWord();
179  m_queryUri = firstLine.eatWord();
180  m_version = firstLine.eatWord();
181 
182  // 验证URI格式（必须以'/'开头）
183  if(m_queryUri.empty() || m_queryUri[0] != '/')
184  {
185  ERROR("Invalid URI: %.*s (must start with '/')", (int)m_queryUri.size(), m_queryUri.data());
186  m_method.clear();
187  m_queryUri.clear();
188  m_version.clear();
189  return Result::Error;
190  }
191 
192  // 解析URI中的查询参数（分割?前后部分）
193  size_t queryPos = m_queryUri.find('?');
194  if(queryPos != std::string::npos)
195  {
196  m_uri = m_queryUri.substr(0, queryPos);
197  Slice queryStr(m_queryUri.data() + queryPos + 1, m_queryUri.size() - queryPos - 1);
198 
199  // 解析键值对（格式：key1=value1&key2=value2&...）
200  size_t pos = 0;
201  while(pos < queryStr.size())
202  {
203  // 查找=的位置
204  size_t eqPos = queryStr.find('=', pos);
205  size_t andPos = queryStr.find('&', pos);
206  size_t keyEnd = Slice::npos;
207 
208  // 确定keyEnd为两者中较小的有效位置
209  if(eqPos != Slice::npos && andPos != Slice::npos)
210  {
211  keyEnd = std::min(eqPos, andPos);
212  }
213  else if(eqPos != Slice::npos)
214  {
215  keyEnd = eqPos;
216  }
217  else if(andPos != Slice::npos)
218  {
219  keyEnd = andPos;
220  }
221  else
222  {
223  // 未找到
224  keyEnd = queryStr.size();
225  }
226 
227  Slice key = queryStr.sub(pos, keyEnd);
228  if(key.empty())
229  {
230  pos = keyEnd + 1;
231  continue;
232  }
233 
234  // 找到value结束位置（&）
235  pos = keyEnd + (keyEnd < queryStr.size() && queryStr[keyEnd] == '=' ? 1 : 0);
236  size_t valEnd = queryStr.find('&', pos);
237  if(valEnd == Slice::npos)
238  valEnd = queryStr.size();
239  Slice val = queryStr.sub(pos, valEnd);
240 
241  // 存入参数映射
242  m_args[std::string(key.data(), key.size())] = std::string(val.data(), val.size());
243  pos = valEnd + 1;
244  }
245  }
246  else
247  {
248  // 无查询参数，URI即为路径
249  m_uri = m_queryUri;
250  }
251  }
252 
253  return result;
254  }

396  void HttpConnPtr::handleRead(const HttpCallBack& cb) const
397  {
398  // 服务端处理请求
399  if(!m_tcp->isClient())
400  {
401  HttpRequest& req = getRequest();
402  HttpMsg::Result result = req.tryDecode(m_tcp->getInputBuffer());
403 
404  if(result == HttpMsg::Result::Error)
405  {
406  m_tcp->close();
407  return;
408  }
409  else if(result == HttpMsg::Result::Continue100)
410  {
411  // 发送100 Continue响应
412  m_tcp->send("HTTP/1.1 100 Continue\r\n\r\n");
413  }
414  else if(result == HttpMsg::Result::Complete)
415  {
416  INFO("Received HTTP request: %s %s %s",
417  req.m_method.c_str(), req.m_uri.c_str(), req.getVersion().c_str());
418  TRACE("Request data:\n%.*s", (int)m_tcp->getInputBuffer().size(), m_tcp->getInputBuffer().begin());
419  // 调用请求处理回调函数
420  cb(*this);
421  }
422  }
423  // 客户端处理响应
424  else
425  {
426  HttpResponse& resp = getResponse();
427  HttpMsg::Result result = resp.tryDecode(m_tcp->getInputBuffer());
428 
429  if(result == HttpMsg::Result::Error)
430  {
431  m_tcp->close();
432  return;
433  }
434  else if(result == HttpMsg::Result::Complete)
435  {
436  INFO("Received HTTP response: %d %s", resp.m_status, resp.m_statusMsg.c_str());
437  TRACE("Response data:\n%.*s", (int)m_tcp->getInputBuffer().size(), m_tcp->getInputBuffer().begin());
438  cb(*this);
439  }
