59  const std::string& uri, const std::string& query = "") {
60  std::thread serverThread([serverBase]() {
61  INFO("Server event loop started");
62  // serverBase->runAfter(2000, [serverBase]{ serverBase->exit(); });
63  serverBase->loop();
64  INFO("Server event loop exited");
65  });
66 
67  std::this_thread::sleep_for(std::chrono::milliseconds(100));
68 
69  EventBase clientBase;
70  std::promise<std::string> promise;
71  auto future = promise.get_future();
72 
73  auto conn = TcpConn::createConnection(&clientBase, "127.0.0.1", 8080);
74  conn->onState([&](const TcpConnPtr& c) {
75  if (c->getState() == TcpConn::State::CONNECTED) {
76  DEBUG("onState: TcpConn::State::CONNECTED");
77  HttpRequest req;
78  req.m_method = "GET";
79  req.m_uri = uri;
80  req.m_queryUri = query.empty() ? uri : uri + "?stat=" + query;
81  Buffer buf;
82  req.encode(buf);
83  c->send(buf);
84  }
85  });
86 
87  conn->onMsg(std::make_unique<LineCodec>(), [&](const TcpConnPtr& c, const Slice& msg) {
88  HttpResponse resp;
89  auto res = resp.tryDecode(msg);
90  if (res == HttpMsg::Result::Complete) {
91  promise.set_value(resp.getBody().toString());
92  c->close();
93  clientBase.exit(); // ✅ 保证客户端 loop 退出
94  }
95  });
96 
97  std::thread clientThread([&]() {
98  INFO("Client event loop started");
99  // clientBase->runAfter(2000, [clientBase]{ clientBase->exit(); });
100  clientBase.loop();
101  INFO("Client event loop exited");
102  });
103 
104  // 等待响应
105  std::string response = future.get();
106 
107  // ✅ 等客户端线程结束
108  clientThread.join();
109 
110  // ✅ 告诉服务器事件循环退出
111  serverBase->exit();
112 
113  // ✅ 等服务器线程结束

309 void testStatServerThreadSafe() {
310  DEBUG("=== 开始测试 StatServer 线程安全 ===");
311 
312  const int THREAD_NUM = 3;
313  const int REQ_PER_THREAD = 10;
314  EventBase base;
315  StatServer server(&base);
316 
317  // 注册测试回调
318  server.onState("thread_safe_state", "线程安全测试状态", []() -> int64_t {
319  return g_callbackInvokeCount.fetch_add(1, std::memory_order_relaxed);
320  });
321 
322  // 启动服务器线程
323  std::thread serverThread([&base]() {
324  base.loop();
325  });
326 
327  // 等待服务器启动
328  std::this_thread::sleep_for(std::chrono::milliseconds(100));
329 
330  // 多线程发送请求
331  std::vector<std::thread> clientThreads;
332  for (int i = 0; i < THREAD_NUM; ++i) {
333  clientThreads.emplace_back([i]() {
334  for (int j = 0; j < REQ_PER_THREAD; ++j) {
335  EventBase clientBase;
336  auto client = TcpConn::createConnection(&clientBase, "127.0.0.1", 8080);
337  std::promise<bool> reqPromise;
338 
339  client->onWritable([&](const TcpConnPtr& conn) {
340  HttpRequest req;
341  req.m_method = "GET";
342  req.m_queryUri = "/thread_safe_state";
343  Buffer buf;
344  req.encode(buf);
345  conn->send(buf);
346  });
347 
348  client->onMsg(std::make_unique<LineCodec>(), [&](const TcpConnPtr& conn, const Slice& msg) {
349  HttpResponse resp;
350  if (resp.tryDecode(msg) == HttpMsg::Result::Complete) {
351  reqPromise.set_value(true);
352  conn->close();
353  clientBase.exit();
354  }
355  });
356  clientBase.loop();
357  reqPromise.get_future().get();
358  }
359  });
360  }
361 
362  // 等待所有客户端线程完成
363  for (auto& t : clientThreads) {
364  t.join();
365  }
366 
367  // 停止服务器
368  base.exit();
369  serverThread.join();
370 
371  // 验证总调用次数
372  int64_t total = g_callbackInvokeCount.load();
373  int64_t expected = THREAD_NUM * REQ_PER_THREAD;
374  bool test1 = (total == expected);
375  DEBUG("测试1（多线程回调计数）：%ld/%ld, %s",
376  total, expected, test1 ? "通过" : "失败");
377 
378  DEBUG("=== StatServer 线程安全测试结束 ===\n");
379 }
