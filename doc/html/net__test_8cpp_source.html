373 void test_buffer_thread_safety() {
374  DEBUG("=== 开始测试Buffer多线程安全 ===");
375  Buffer buf;
376  const int THREAD_NUM = 5; // 5个并发线程
377  const int LOOP_NUM = 500; // 每个线程循环500次
378 
379  // 线程1-2：并发追加字符串
380  auto appendThread = [&buf]() {
381  for (int i = 0; i < LOOP_NUM; ++i) {
382  std::string data = "thread_" + std::to_string(pthread_self()) + "_" + std::to_string(i);
383  DEBUG("线程%lu追加数据: %s", pthread_self(), data.c_str());
384  buf.append(data);
385  g_bufferThreadCount.fetch_add(1, std::memory_order_relaxed);
386  }
387  };
388 
389  // 线程3-4：并发消耗数据（每次消耗1-5字节）
390  auto consumeThread = [&buf]() {
391  for (int i = 0; i < LOOP_NUM; ++i) {
392  size_t currentSize = buf.size();
393  if (currentSize > 0) {
394  size_t consumeLen = std::min<size_t>(rand() % 5 + 1, currentSize);
395  DEBUG("线程%lu消耗数据: %zu字节", pthread_self(), consumeLen);
396  buf.consume(consumeLen);
397  }
398  g_bufferThreadCount.fetch_add(1, std::memory_order_relaxed);
399  }
400  };
401 
402  // 线程5：并发吸收临时缓冲区
403  auto absorbThread = [&buf]() {
404  for (int i = 0; i < LOOP_NUM; ++i) {
405  Buffer tempBuf;
406  {
407  tempBuf.append("absorb_temp_" + std::to_string(i));
408  }
409  DEBUG("线程%lu吸收临时缓冲区: %s", pthread_self(), tempBuf.data().c_str());
410  buf.absorb(tempBuf);
411  g_bufferThreadCount.fetch_add(1, std::memory_order_relaxed);
412  }
413  };
414 
415  // 启动线程
416  std::vector<std::thread> threads;
417  threads.emplace_back(appendThread);
418  threads.emplace_back(appendThread);
419  threads.emplace_back(consumeThread);
420  threads.emplace_back(consumeThread);
421  threads.emplace_back(absorbThread);
422 
423  // 等待所有线程完成
424  for (auto& t : threads) {
425  t.join();
426  }
427 
428  // 验证：无崩溃+执行次数正确
429  int64_t total = g_bufferThreadCount.load();
430  int64_t expected = THREAD_NUM * LOOP_NUM;
431  bool threadSafe = (total == expected);
432  DEBUG("多线程执行次数测试: 实际=%lld，预期=%lld（%s）",
433  total, expected, threadSafe ? "通过" : "失败");
434  DEBUG("多线程后缓冲区状态: 非空=%s，长度=%zu（无崩溃即符合安全要求）",
435  buf.empty() ? "否" : "是", buf.size());
436 
437  DEBUG("=== Buffer多线程安全测试结束 ===\n");
438 }
