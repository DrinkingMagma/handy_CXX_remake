276  Slice eatLine() noexcept {
277  const char* p = m_pb; // 记录行起始位置
278  const char* end = m_pb; // 记录行结束位置（不含换行符）
279 
280  // 查找换行符，同时记录行内容的结束位置
281  while (m_pb < m_pe) {
282  if (*m_pb == '\n' || *m_pb == '\r') {
283  end = m_pb; // 行内容到换行符前结束
284  // 跳过当前换行符
285  ++m_pb;
286  // 处理Windows换行（\r\n）：若当前是'\n'且前一个是'\r'，再跳过'\n'
287  if (m_pb < m_pe && *m_pb == '\n' && (m_pb - 1 >= p && *(m_pb - 1) == '\r')) {
288  ++m_pb;
289  }
290  break; // 找到换行符，退出循环
291  } else {
