206 // 测试多线程并发访问
207 void test_udp_thread_safe() {
208  DEBUG("=== 开始测试 UDP 线程安全 ===");
209 
210  const int THREAD_NUM = 5;
211  const int MSG_PER_THREAD = 20;
212  EventBase* base = new EventBase();
213  std::atomic<int> msgCount(0);
214 
215  // 启动服务器
216  auto server = UdpServer::startServer(base, "127.0.0.1", 12348);
217  if (!server) {
218  ERROR("UdpServer 启动失败");
219  return;
220  }
221 
222  server->onMsg([&](const UdpServer::Ptr& srv, Buffer buf, Ipv4Addr peer) {
223  msgCount++;
224  srv->sendTo("ACK", peer);
225  });
226 
227  // 启动多个客户端线程
228  auto clientFunc = [&]() {
229  EventBase clientBase;
230  auto conn = UdpConn::createConnection(&clientBase, "127.0.0.1", 12348);
231  if (!conn) return;
232 
233  int ackCount = 0;
234  conn->onMsg([&](const UdpConn::Ptr&, Buffer) {
235  if (++ackCount >= MSG_PER_THREAD) {
236  clientBase.exit();
237  }
238  });
239 
240  for (int i = 0; i < MSG_PER_THREAD; ++i) {
241  conn->send(g_testMsg + std::to_string(i));
242  std::this_thread::sleep_for(std::chrono::microseconds(100));
243  }
244 
245  clientBase.runAfter(1000, [&]() { clientBase.exit(); });
246  clientBase.loop();
247  g_udpTestCount += ackCount;
248  };
249 
250  std::vector<std::thread> clients;
251  for (int i = 0; i < THREAD_NUM; ++i) {
252  clients.emplace_back(clientFunc);
253  }
254 
255  // 运行服务器
256  base->runAfter(2000, [&]() { base->exit(); });
257  base->loop();
258 
259  // 等待所有客户端完成
260  for (auto& t : clients) {
261  t.join();
262  }
263 
264  int expected = THREAD_NUM * MSG_PER_THREAD;
265  bool testOk = (msgCount == expected) && (g_udpTestCount == expected);
266  DEBUG("测试1（多线程并发）：总接收=%d, 总确认=%d, 预期=%d, %s",
267  (int)msgCount, (int)g_udpTestCount, expected, testOk ? "通过" : "失败");
268  DEBUG("=== UDP 线程安全测试结束 ===\n");
269 }
