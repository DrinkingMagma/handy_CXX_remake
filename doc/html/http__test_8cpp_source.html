390 void testHttpThreadSafe() {
391  DEBUG("=== 开始测试 HTTP 线程安全 特性 ===");
392 
393  const int THREAD_NUM = 5; // 5个测试线程
394  const int LOOP_NUM = 100; // 每个线程循环100次
395  std::vector<std::thread> threads;
396 
397  // 线程函数：循环创建和处理HTTP消息
398  auto thread_func = []() {
399  for (int i = 0; i < LOOP_NUM; ++i) {
400  // 测试请求
401  HttpRequest req;
402  req.m_method = "GET";
403  req.m_queryUri = "/thread/test?i=" + std::to_string(i);
404  req.setHeader("Thread-Id", std::to_string(std::hash<std::thread::id>()(std::this_thread::get_id())));
405 
406  // 测试响应
407  HttpResponse resp;
408  resp.setStatus(200);
409  resp.setBody("Response from thread");
410 
411  // 编码测试
412  Buffer buf;
413  req.encode(buf);
414  buf.clear();
415  resp.encode(buf);
416 
417  g_httpThreadTestCount.fetch_add(1, std::memory_order_relaxed);
418  }
419  };
420 
421  // 创建线程
422  for (int i = 0; i < THREAD_NUM; ++i) {
423  threads.emplace_back(thread_func);
424  }
425 
426  // 等待所有线程结束
427  for (auto& t : threads) {
428  t.join();
429  }
430 
431  // 验证总执行次数
432  int64_t total = g_httpThreadTestCount.load();
433  int64_t expected = THREAD_NUM * LOOP_NUM;
434  bool thread_ok = (total == expected);
435  DEBUG("线程安全测试：总执行次数=%lld（预期：%lld，%s）",
