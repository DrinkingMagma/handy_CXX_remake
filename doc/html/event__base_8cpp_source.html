382  TimerId runAt(int64_t timestamp_ms, Task&& task, int64_t interval_ms)
383  {
384  // 已退出或任务为空，返回无效ID
385  if(m_exit || !task)
386  return TimerId();
387 
388  // 处理可重复定时器
389  if(interval_ms > 0)
390  {
391  // 生成可重复定时器的自定义ID（first为负数，区分于一次性定时器）
392  TimerId rep{-timestamp_ms, ++m_timerSeq};
393  auto [it, inserted] = m_timerReps.emplace(
394  rep, TimerRepeatable{
395  timestamp_ms, interval_ms, {timestamp_ms, ++m_timerSeq}, std::move(task)});
396  if(!inserted)
397  return TimerId();
398 
399  TimerRepeatable* tr = &it->second;
400  // 注册当前周期的一次性定时器
401  m_timers[tr->timerIdPair] = [this, tr]() { repeatableTimeout(tr); };
402  refreshNearestTimer(&tr->timerIdPair);
403 
404  TRACE("Repeatable timer registered: repTimerIdPair={%lld, %lld}, interval=%lld ms",
405  rep.first, rep.second, interval_ms);
406  return rep;
407  }
408 
409  TimerId tid{timestamp_ms, ++m_timerSeq};
410  m_timers.emplace(tid, std::move(task));
411  refreshNearestTimer(&tid);
412 
413  TRACE("One-shot timer registered: timerIdPair={%lld, %lld}",
414  tid.first, tid.second);
415  return tid;
