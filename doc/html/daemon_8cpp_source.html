293  {
294  if(!pidFilePath || strlen(pidFilePath) == 0)
295  {
296  fprintf(stderr, "Pid file path is invalid\n");
297  return -1;
298  }
299 
300  int pid = getPidFromFile(pidFilePath);
301  if(pid <= 0)
302  {
303  fprintf(stderr, "No such daemon process, pid file(%s) is empty or not exist or contains invalid pid\n", pidFilePath);
304  return -1;
305  }
306 
307  // 尝试优雅终止
308  if(kill(pid, SIGTERM) < 0)
309  {
310  if(errno == ESRCH)
311  {
312  fprintf(stderr, "No such daemon process, pid=%d, removing pid file\n", pid);
313  unlink(pidFilePath);
314  return 0;
315  }
316  fprintf(stderr, "Send SIGTERM to kill daemon process(pid=%d) failed: errno=%d, msg=%s\n", pid, errno, strerror(errno));
317  return -1;
318  }
319 
320  // 等待进程退出，最多等待3秒
321  for(int i = 0; i < 300; ++i)
322  {
323  usleep(10 * 1000); // 10ms
324  if(kill(pid, 0) < 0)
325  {
326  // 若进程已退出
327  if(errno == ESRCH)
328  {
329  fprintf(stderr, "Daemon process(pid=%d) has been killed\n", pid);
330  unlink(pidFilePath);
331  return 0;
332  }
333  fprintf(stderr, "Check daemon process(pid=%d) status failed: errno=%d, msg=%s\n", pid, errno, strerror(errno));
334  return -1;
335  }
336  }
337 
338  // 优雅终止失败，尝试强制终止
339  fprintf(stderr, "Graceful stop daemon process(pid=%d) failed, trying SIGKILL\n", pid);
340  if(kill(pid, SIGKILL) < 0)
341  {
342  fprintf(stderr, "Send SIGKILL to kill daemon process(pid=%d) failed: errno=%d, msg=%s\n", pid, errno, strerror(errno));
343  return -1;
344  }

505  void Signal::signal(int sig, const std::function<void()>& handler)
506  {
507  std::lock_guard<std::mutex> lock(signalHandlersMutex);
508 
509  // 保存新的信号处理函数
510  signalHandlers[sig] = handler;
511 
512  // 设置信号处理函数
513  struct sigaction sa;
514  std::memset(&sa, 0, sizeof(sa));
515  sa.sa_handler = &Signal::_signalHandler;
516  sigemptyset(&sa.sa_mask);
517  // 不使用SA_RESTART，避免中断系统调用后自动重启
518  sa.sa_flags = 0;
519 
520  if(sigaction(sig, &sa, nullptr) < 0)
521  {
522  fprintf(stderr, "Set signal handler for signal(%d) failed: errno=%d, msg=%s\n", sig, errno, strerror(errno));
523  signalHandlers.erase(sig);
524  }
525  }
