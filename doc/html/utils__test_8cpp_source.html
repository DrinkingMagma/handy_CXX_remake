221 void test_thread_safe() {
222  DEBUG("=== 开始测试 线程安全 特性 ===");
223 
224  const int THREAD_NUM = 5; // 5个测试线程
225  const int LOOP_NUM = 100; // 每个线程循环100次
226  std::vector<std::thread> threads;
227 
228  // 线程函数：循环调用readableTime和format（高频操作验证竞争）
229  auto thread_func = []() {
230  for (int i = 0; i < LOOP_NUM; ++i) {
231  time_t now = time(nullptr);
232  std::string time_str = Utils::readableTime(now);
233  std::string fmt_str = Utils::format("线程%d-%d：time=%s",
234  std::this_thread::get_id(), i, time_str.c_str());
235  // 日志输出（间接验证无崩溃/乱码）
236  DEBUG("%s", fmt_str.c_str());
237  g_threadTestCount.fetch_add(1, std::memory_order_relaxed);
238  }
239  };
240 
241  // 创建线程
242  for (int i = 0; i < THREAD_NUM; ++i) {
243  threads.emplace_back(thread_func);
244  }
245 
246  // 等待所有线程结束
247  for (auto& t : threads) {
248  t.join();
249  }
250 
251  // 验证总执行次数（确保无线程遗漏）
252  int64_t total = g_threadTestCount.load();
253  int64_t expected = THREAD_NUM * LOOP_NUM;
254  bool thread_ok = (total == expected);
255  DEBUG("线程安全测试：总执行次数=%lld（预期：%lld，%s）",
256  total, expected, thread_ok ? "通过" : "失败");
257  DEBUG("=== 线程安全 特性测试结束 ===\n");
258 }
