247  void Logger::checkAndRotateLogFile()
248  {
249  int64_t now = getCurrentTimeStamp();
250  // load(std::memory_order_acquire)：原子地读取该变量的值
251  // std::memory_order_acquire 是内存序参数，确保后续操作不会被编译器或 CPU 重排到该读取操作之前，保证多线程下的内存可见性。
252  int64_t lastRotateTime = m_realRotateTime.load(std::memory_order_acquire);
253 
254  // 检查是否需要轮转
255  bool isNeedRotate = false;
256 
257  // 时间触发
258  if(now - lastRotateTime >= m_rotateInterval)
259  {
260  isNeedRotate = true;
261  }
262  // 文件大小触发
263  else if(m_logFileName.empty() == false)
264  {
265  std::lock_guard<std::mutex> lock(m_mutex);
266  if(m_currentFileSize >= m_maxLogFileSize)
267  {
268  isNeedRotate = true;
269  }
270  }
271 
272  // 尝试进行轮转
273  // m_realRotateTime.compare_exchange_strong(lastRotateTime, now)：原子比较并交换操作（CAS）
274  // 如果 m_realRotateTime 当前的值等于 lastRotateTime（即上次读取的旧值，期间未被其他线程修改），则将其更新为 now（当前时间），返回 true；否则返回 false。
275  // 避免多线程同时触发轮转（例如两个线程同时判断需要轮转，CAS 确保只有一个线程能成功执行后续操作）
276  if(isNeedRotate && m_realRotateTime.compare_exchange_strong(lastRotateTime, now))
277  {
278  std::lock_guard<std::mutex> lock(m_mutex);
279  rotateLogFile();
280  m_lastRotateTime = now;
281  }
282  }

310  fprintf(stderr, "Failed to rotate log file: %s -> %s, because rename file failed!\n",
311  m_logFileName.c_str(), backupFileName.c_str());
312  }
313 
314  // 打开新的日志文件
315  m_fd = fopen(m_logFileName.c_str(), "a");
316  if(!m_fd)
317  {
318  fprintf(stderr, "Failed to open new log file: %s\n", m_logFileName.c_str());
