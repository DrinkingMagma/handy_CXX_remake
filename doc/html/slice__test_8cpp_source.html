230 void test_thread_safe() {
231  DEBUG("=== 开始测试 Slice 线程安全 特性 ===");
232 
233  const int THREAD_NUM = 5; // 5个测试线程
234  const int LOOP_NUM = 100; // 每个线程循环100次
235  std::vector<std::thread> threads;
236 
237  // 线程函数：循环创建和访问Slice（高频操作验证竞争）
238  auto thread_func = []() {
239  for (int i = 0; i < LOOP_NUM; ++i) {
240  Slice s("hello world");
241  std::string str = s.toString();
242 
243  auto tid = std::this_thread::get_id();
244  std::stringstream ss;
245  ss << tid;
246  DEBUG("线程%s-%d：slice=%s", ss.str().c_str(), i, str.c_str());
247  g_sliceThreadTestCount.fetch_add(1, std::memory_order_relaxed);
248  }
249  };
250 
251  // 创建线程
252  for (int i = 0; i < THREAD_NUM; ++i) {
253  threads.emplace_back(thread_func);
254  }
255 
256  // 等待所有线程结束
257  for (auto& t : threads) {
258  t.join();
259  }
260 
261  // 验证总执行次数（确保无线程遗漏）
262  int64_t total = g_sliceThreadTestCount.load();
263  int64_t expected = THREAD_NUM * LOOP_NUM;
264  bool thread_ok = (total == expected);
265  DEBUG("线程安全测试：总执行次数=%ld（预期：%ld，%s）",
266  total, expected, thread_ok ? "通过" : "失败");
267  DEBUG("=== Slice 线程安全 特性测试结束 ===\n");
268 }
