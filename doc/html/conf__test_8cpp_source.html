274 void test_thread_safety() {
275  DEBUG("=== 开始测试线程安全特性 ===");
276 
277  const std::string iniContent = R"( [ThreadTest] Value1 = 100 Value2 = 3.14 Value3 = true Value4 = test_string )";
278  const std::string tempFile = "thread_test.ini";
279  createTempIniFile(iniContent, tempFile);
280 
281  // 加载配置
282  Conf conf;
283  conf.parse(tempFile);
284 
285  const int THREAD_NUM = 4; // 4个测试线程
286  const int LOOP_NUM = 1000; // 每个线程循环1000次
287 
288  // 线程函数: 并发读取配置
289  auto thread_func = [&conf]() {
290  for (int i = 0; i < LOOP_NUM; ++i) {
291  // 混合调用各种获取方法
292  conf.get("ThreadTest", "Value4", "");
293  conf.getInteger("ThreadTest", "Value1", 0);
294  conf.getReal("ThreadTest", "Value2", 0.0);
295  conf.getBoolean("ThreadTest", "Value3", false);
296  g_threadTestCount.fetch_add(1, std::memory_order_relaxed);
297  }
298  };
299 
300  // 创建并启动线程
301  std::vector<std::thread> threads;
302  for (int i = 0; i < THREAD_NUM; ++i) {
303  threads.emplace_back(thread_func);
304  }
305 
306  // 等待所有线程完成
307  for (auto& t : threads) {
308  t.join();
309  }
310 
311  // 验证执行次数
312  int64_t total = g_threadTestCount.load();
313  int64_t expected = THREAD_NUM * LOOP_NUM;
314  bool threadOk = (total == expected);
315  DEBUG("线程安全测试: 总执行次数=%ld（预期: %ld，%s）",
316  total, expected, threadOk ? "通过" : "失败");
317 
318  deleteTempIniFile(tempFile);
319  DEBUG("=== 线程安全特性测试结束 ===\n");
320 }
