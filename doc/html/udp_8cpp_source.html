48  int UdpServer::bind(const std::string& host, unsigned short port, bool isReusePort)
49  {
50  m_addr = Ipv4Addr(host, port);
51 
52  // 创建UDP套接字
53  int fd = socket(AF_INET, SOCK_DGRAM, 0);
54  if(fd < 0)
55  {
56  ERROR("Create socket failed: errno=%d, msg=%s", errno, strerror(errno));
57  return errno;
58  }
59 
60  // 设置地址复用
61  int r = Net::setReuseAddr(fd, true);
62  if(r != 1)
63  {
64  ERROR("Set reuse addr failed: errno=%d, msg=%s", errno, strerror(errno));
65  close(fd);
66  return errno;
67  }
68 
69  // 设置端口复用
70  if(isReusePort)
71  {
72  r = Net::setReusePort(fd, true);
73  if(r != 1)
74  {
75  ERROR("Set reuse port failed: errno=%d, msg=%s", errno, strerror(errno));
76  close(fd);
77  return errno;
78  }
79  }
80 
81  // 设置FD_CLOEXEC标志
82  r = Utils::addFdFlag(fd, FD_CLOEXEC);
83  if(r != 0)
84  {
85  ERROR("Set FD_CLOEXEC failed: errno=%d, msg=%s", errno, strerror(errno));
86  close(fd);
87  return errno;
88  }
89 
90  // 绑定地址
91  struct sockaddr_in sockAddr = m_addr.getAddr();
92  r = ::bind(fd, reinterpret_cast<struct sockaddr*>(&sockAddr), sizeof(sockAddr));
93  if(r != 0)
94  {
95  ERROR("Bind to %s failed: errno=%d, msg=%s", m_addr.toString().c_str(), errno, strerror(errno));
96  close(fd);
97  return errno;
98  }
99 
100  // 设置非阻塞模式
101  if(Net::setNonBlock(fd) != 1)
102  {
103  ERROR("Set non-block failed: errno=%d, msg=%s", errno, strerror(errno));
104  close(fd);
105  return errno;
106  }
107 
108  INFO("UDP server(fd=%d) bind to %s success", fd, m_addr.toString().c_str());
109 
110  // 创建通道并设置读事件回调：读取数据并调用消息处理回调函数
111  m_channel = new Channel(m_base, fd, kReadEvent);
112  m_channel->onRead([this]() {
113  if(!m_channel || m_channel->getFd() < 0)
114  return;
115 
116  Buffer buf;
117  struct sockaddr_in remoteAddr;
118  socklen_t remoteAddrLen = sizeof(remoteAddr);
119  int fd = m_channel->getFd();
120 
121  TRACE("Udp server(fd=%d) recving...", fd);
122  while(true)
123  {
124  ssize_t rn = recvfrom(fd, buf.makeRoom(kUdpPacketSize), kUdpPacketSize, 0,
125  reinterpret_cast<struct sockaddr*>(&remoteAddr), &remoteAddrLen);
126 
127  if(rn > 0)
128  {
129  buf.addSize(static_cast<size_t>(rn));
130  TRACE("Udp server(fd=%d) recvfrom %ld bytes from %s", fd, rn, Ipv4Addr(remoteAddr).toString().c_str());
131  TRACE("Udp server(fd=%d) recvfrom %s", fd, buf.data().c_str());
132 
133  if(m_serverMsgCallback)
134  {
135  m_serverMsgCallback(shared_from_this(), buf, Ipv4Addr(remoteAddr));
136  }
137  }
138  // 没有更多数据，退出循环
139  else if(rn < 0 && (errno == EAGAIN || errno == EWOULDBLOCK))
140  {
141  break;
142  }
143  else if(rn < 0 && errno != EINTR)
144  {
145  ERROR("UDP server(fd=%d) recvfrom failed: errno=%d, msg=%s", fd, errno, strerror(errno));
146  break;
147  }
148  else
149  break;
150  }
151  });
152 
153  return 0;
154  }
