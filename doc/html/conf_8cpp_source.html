226  int Conf::parse(const std::string& fileName)
227  {
228  // 记录当前解析的文件名
229  m_fileName = fileName;
230  // 以只读模式打开
231  FILE *file = fopen(m_fileName.c_str(), "r");
232  if(!file)
233  return -1;
234 
235  // 使用智能指针自动关闭文件（避免资源泄露）
236  auto autoClose = std::shared_ptr<FILE>(file, fclose);
237 
238  // 最大行长度（16KB）
239  static const int MAX_LINE = 16 * 1024;
240  // 行缓冲区,智能指针自动释放缓冲区
241  auto lineBuf = std::make_unique<char[]>(MAX_LINE);
242 
243  int curLine = 0; // 当前行号
244  string section; // 当前解析的节名称
245  Slice key; // 当前解析的键名称
246  int err = 0; // 错误标志
247 
248  // 逐行读取文件
249  while(!err && fgets(lineBuf.get(), MAX_LINE, file) != nullptr)
250  {
251  curLine++;
252  // 初始化行扫描器
253  LineScanner scanner(lineBuf.get());
254  // 获取行首的第一个非空格字符
255  int firstChar = scanner.peekChar();
256 
257  // 处理注释行或空行（跳过）
258  if(firstChar == ';' || firstChar == '#' || firstChar == '\0')
259  continue;
260  // 处理节定义，如：[section]
261  else if(firstChar == '[')
262  {
263  section = scanner.skip(1).consumeTill(']');
264  err = scanner.match(']').err;
265  // 新节开始，重置当前键名
266  key.clear();
267  }
268  // 处理续行（以空格开头，说明属于上一个键的值）
269  else if (isspace(static_cast<unsigned char>(lineBuf[0])))
270  {
271  if(!key.empty())
272  {
273  // 将续行的内容添加到上一个键的值列表中
274  Slice val = scanner.consumeTill('\0');
275  val.trimSpace();
276  m_values[makeKey(section, key)].push_back(val);
277  }
278  else
279  err = 1;
280  }
281  // 处理键值对（如key = value或key::value）
282  else
283  {
284  // 备份扫描器状态（用于兼容:分隔符）
285  LineScanner backUp = scanner;
286  // 尝试以'='分隔键和值
287  key = scanner.consumeTill('=');
288  key.trimSpace();
289 
290  if(scanner.peekChar() == '=')
291  scanner.skip(1);
292  else
293  {
294  scanner = backUp;
295  key = scanner.consumeTill(':');
296  err = scanner.match(':').err;
297  }
298  scanner.skipSpace();
299  // 提取键值对并存储
300  Slice val = scanner.consumeTillEnd();
301  val.trimSpace();
302  m_values[makeKey(section, key)].push_back(val);
303 
304  // std::cout << section << ": " << "-" << key.toString() << "- = -" << val.toString() << "-" << std::endl;
305  }
306  }
307 
308  return err ? curLine : 0;
309  }
