47  StatServer::StatServer(EventBase* base) : m_server(base)
48  {
49  m_server.onDefault([this](const HttpConnPtr& conn) {
50  HttpRequest& req = conn.getRequest();
51  HttpResponse& resp = conn.getResponse();
52  Buffer buf;
53 
54  // 解析查询参数（优先从?stat=xxx获取，其次从URI路径获取）
55  std::string query = req.getArg("stat");
56  if(query.empty() && !req.m_uri.empty() && req.m_uri != "/")
57  {
58  // 去除URI中的开头'/'
59  query = req.m_uri.substr(1);
60  }
61 
62  // 处理具体查询
63  if(!query.empty())
64  {
65  auto it = m_allCallBacks.find(query);
66  if(it != m_allCallBacks.end())
67  {
68  // 调用注册的回调函数
69  it->second(req, resp);
70  }
71  else
72  {
73  // 未找到对应处理函数
74  resp.setNotFound();
75  }
76  }
77 
78  // 根路径显示汇总页面
79  if(req.m_uri == "/")
80  {
81  // 刷新按钮
82  buf.append("<a href=\"/\">刷新</a><br/>\n");
83 
84  // 状态信息表格
85  buf.append("<table border=\"1\">\n");
86  buf.append("<tr><th>状态<</th><th>描述</th><th>数量</th></tr>\n");
87  for(const auto& entry : m_statCallBacks)
88  {
89  HttpResponse tempResp;
90  HttpRequest tempReq;
91  tempReq.m_uri = entry.first;
92  entry.second.callback(tempReq, tempResp);
93 
94  buf.append("<tr><td>")
95  .append(pageLink(entry.first))
96  .append("</td><td>")
97  .append(entry.second.description)
98  .append("</td><td>")
99  .append(tempResp.getBody())
100  .append("</td></tr>\n");
101  }
102  buf.append("</table>\n<br/>\n");
103 
104  // 页面信息表格
105  buf.append("<table border=\"1\">\n");
106  buf.append("<tr><th>页面</th><th>描述</th></tr>\n");
107  for(const auto& entry : m_pageCallBacks)
108  {
109  buf.append("<tr><td>")
110  .append(pageLink(entry.first))
111  .append("</td><td>")
112  .append(entry.second.description)
113  .append("</td></tr>\n");
114  }
115  buf.append("</table>\n<br/>\n");
116 
117  // 命令信息表格
118  buf.append("<table border=\"1\">\n");
119  buf.append("<tr><th>命令</th><th>描述</th></tr>\n");
120  for(const auto& entry : m_cmdCallBacks)
121  {
122  buf.append("<tr><td>")
123  .append(queryLink(entry.first))
124  .append("</td><td>")
125  .append(entry.second.description)
126  .append("</td></tr>\n");
127  }
128  buf.append("</table>\n");
129 
130  // 添加子查询结果（如果有）
131  if(!resp.getBody().empty())
132  {
133  buf.append(Utils::format("<br/>子查询 %s 结果:<br/>%s", query.c_str(), resp.getBody().toString().c_str()));
134  }
135 
136  resp.setBody(buf.data());
137  resp.setHeader("Content-Type", "text/html; charset=utf-8");
138  }
139 
140  // 发送响应并记录日志
141  INFO("Response status: %d, content lenth: %zu", resp.m_status, resp.getBody().size());
142  conn.sendResponse();
143  });
144  }

146  void StatServer::onRequest(StatType type, const std::string& key, const std::string& desc, const StatCallBack& cb)
147  {
148  if(key.empty())
149  {
150  ERROR("register failed: key can not be empty");
151  return;
152  }
153 
154  // 根据类型存储回调
155  CallBackInfo info{desc, cb};
156  switch(type)
157  {
158  case StatType::STATE:
159  m_statCallBacks[key] = info;
160  break;
161  case StatType::PAGE:
162  m_pageCallBacks[key] = info;
163  break;
164  case StatType::CMD:
165  m_cmdCallBacks[key] = info;
166  break;
167  default:
168  ERROR("Unknown StatType: %d", static_cast<int>(type));
169  return;
170  }
171  // 加入全局映射表
172  m_allCallBacks[key] = cb;
173  }

185  void StatServer::onPageFile(const std::string& page, const std::string& desc, const std::string& file)
186  {
187  onRequest(StatType::PAGE, page, desc, [file](const HttpRequest&, HttpResponse& resp){
188  std::string content;
189  Status st = File::getContent(file, content);
190  if(!st.ok())
191  {
192  ERROR("get file content failed: %s, msg: %s", file.c_str(), st.toString().c_str());
193  resp.setNotFound();
194  return;
195  }
196 
197  resp.setBody(content);
198  // 根据文件后缀设置Content-Type（简化版）
199  if(file.size() >= 5 && file.substr(file.size() - 4) == ".html")
200  {
201  resp.setHeader("Content-Type", "text/html; charset=utf-8");
202  }
203  else if(file.size() >= 4 && file.substr(file.size() - 3) == ".js")
204  {
205  resp.setHeader("Content-Type", "application/javascript; charset=utf-8");
206  }
207  else
208  {
209  resp.setHeader("Content-Type", "text/plain; charset=utf-8");
210  }
211  });
212  }
